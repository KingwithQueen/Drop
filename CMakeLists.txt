cmake_minimum_required(VERSION 3.24)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/dropcommon/version.h" version_line REGEX "^#define DROP_VERSION \"")
string(REGEX MATCH "[0-9]+\.[0-9]+\.[0-9]+" DROP_PRODUCT_VERSION "${version_line}")
project(Drop VERSION ${DROP_PRODUCT_VERSION} LANGUAGES C)
if(WIN32)
    message(FATAL_ERROR "Windows uses drop.slnx and scripts/Build.ps1; this entry builds Linux and macOS.")
endif()
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(CJSON REQUIRED IMPORTED_TARGET libcjson>=1.7.19)
pkg_check_modules(SODIUM REQUIRED IMPORTED_TARGET libsodium>=1.0.18)
pkg_check_modules(SQLITE REQUIRED IMPORTED_TARGET sqlite3>=3.35)
pkg_check_modules(ZSTD REQUIRED IMPORTED_TARGET libzstd>=1.5)
pkg_check_modules(OPENSSL REQUIRED IMPORTED_TARGET openssl>=3.0)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl>=7.86)
add_library(dropcommon STATIC dropcommon/common.c dropcommon/config.c dropcommon/object.c dropcommon/pake.c dropcommon/srp_vendor.c
    platform/encoding.c platform/posix.c platform/storage.c)
target_include_directories(dropcommon PUBLIC dropcommon)
target_compile_definitions(dropcommon PUBLIC _POSIX_C_SOURCE=200809L _DEFAULT_SOURCE DROP_DIAGNOSTICS=0)
target_link_libraries(dropcommon PUBLIC PkgConfig::CJSON PkgConfig::SODIUM PkgConfig::ZSTD PkgConfig::OPENSSL PkgConfig::CURL Threads::Threads)
if(APPLE)
    target_compile_definitions(dropcommon PUBLIC _DARWIN_C_SOURCE)
    target_sources(dropcommon PRIVATE platform/secret_macos.c)
    target_link_libraries(dropcommon PUBLIC "-framework Security" "-framework CoreFoundation")
else()
    pkg_check_modules(SECRET REQUIRED IMPORTED_TARGET libsecret-1)
    target_sources(dropcommon PRIVATE platform/secret_linux.c)
    target_link_libraries(dropcommon PUBLIC PkgConfig::SECRET)
endif()
add_library(dropclient STATIC drop/client.c drop/profiles.c drop/receipts.c drop/deletions.c drop/transfers.c drop/pake_client.c drop/ui_data.c platform/network_curl.c)
target_include_directories(dropclient PUBLIC drop)
target_link_libraries(dropclient PUBLIC dropcommon PkgConfig::SQLITE)
add_library(dropruntime STATIC platform/runtime.c)
target_include_directories(dropruntime PUBLIC platform)
target_link_libraries(dropruntime PUBLIC dropclient)
add_executable(drop platform/main.c)
target_link_libraries(drop PRIVATE dropruntime)
if(APPLE)
    target_sources(drop PRIVATE platform/ui_macos.c)
    target_link_libraries(drop PRIVATE "-framework AppKit" "-framework CoreGraphics" "-framework UserNotifications" objc)
else()
    pkg_check_modules(GTK4 REQUIRED IMPORTED_TARGET gtk4>=4.10)
    target_sources(drop PRIVATE platform/ui_gtk.c)
    target_link_libraries(drop PRIVATE PkgConfig::GTK4)
    find_program(GLIB_COMPILE_RESOURCES glib-compile-resources REQUIRED)
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/drop-resources.c"
        COMMAND "${GLIB_COMPILE_RESOURCES}" "${CMAKE_CURRENT_SOURCE_DIR}/platform/drop.gresource.xml" --generate-source
            "--sourcedir=${CMAKE_CURRENT_SOURCE_DIR}" "--target=${CMAKE_CURRENT_BINARY_DIR}/drop-resources.c"
        DEPENDS platform/drop.gresource.xml platform/drop-style.css Dropico.png)
    target_sources(drop PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/drop-resources.c")
    set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/drop-resources.c" PROPERTIES COMPILE_OPTIONS "-Wno-overlength-strings")
endif()
foreach(target dropcommon dropclient dropruntime drop)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
set_source_files_properties(dropcommon/srp_vendor.c PROPERTIES COMPILE_OPTIONS "-Wno-deprecated-declarations;-Wno-unused-parameter")
# 本机验收文件不进入源码发布；完整保留这些文件时默认启用测试，发布源码默认只构建产品。
set(DROP_LOCAL_CHECKS_AVAILABLE ON)
set(DROP_GUI_CHECK_SOURCE checks/gtk_probe.inc)
if(APPLE)
    set(DROP_GUI_CHECK_SOURCE checks/macos_probe_c.inc)
endif()
foreach(check_source checks/portable_check.c checks/client_fixture.c ${DROP_GUI_CHECK_SOURCE})
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${check_source}")
        set(DROP_LOCAL_CHECKS_AVAILABLE OFF)
    endif()
endforeach()
option(BUILD_TESTING "Build local verification tools excluded from source releases" ${DROP_LOCAL_CHECKS_AVAILABLE})
if(BUILD_TESTING AND NOT DROP_LOCAL_CHECKS_AVAILABLE)
    message(FATAL_ERROR "Local verification sources are excluded from source releases. Configure with -DBUILD_TESTING=OFF.")
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(drop-portable-check checks/portable_check.c)
    target_link_libraries(drop-portable-check PRIVATE dropclient)
    target_compile_options(drop-portable-check PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME portable-core COMMAND drop-portable-check)
    add_executable(drop-client-fixture checks/client_fixture.c)
    target_link_libraries(drop-client-fixture PRIVATE dropruntime)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/checks/sync_performance.c")
        add_executable(drop-sync-performance checks/sync_performance.c)
        target_link_libraries(drop-sync-performance PRIVATE dropruntime)
        target_compile_options(drop-sync-performance PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_executable(drop-gui-check platform/main.c)
    target_link_libraries(drop-gui-check PRIVATE dropruntime)
    target_compile_definitions(drop-gui-check PRIVATE DROP_GUI_CHECKS=1)
    if(APPLE)
        target_sources(drop-gui-check PRIVATE platform/ui_macos.c)
        target_link_libraries(drop-gui-check PRIVATE "-framework AppKit" "-framework CoreGraphics" "-framework UserNotifications" objc)
    else()
        target_sources(drop-gui-check PRIVATE platform/ui_gtk.c)
        target_sources(drop-gui-check PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/drop-resources.c")
        target_link_libraries(drop-gui-check PRIVATE PkgConfig::GTK4)
    endif()
endif()
install(TARGETS drop RUNTIME DESTINATION bin)
